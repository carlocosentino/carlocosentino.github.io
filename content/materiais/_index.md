        (function(){
          document.addEventListener('click', function(e){
            var btn = e.target.closest('.mat-view');
            if(!btn) return;
            var item = btn.closest('.mat-item');
            var viewer = item.nextElementSibling;
            if(!viewer || !viewer.classList.contains('mat-viewer')) return;
            var open = viewer.classList.toggle('is-open');
            if(open && !viewer.querySelector('iframe')){
              var f = document.createElement('iframe');
              f.src = btn.getAttribute('data-src');
              f.loading = 'lazy';
              viewer.appendChild(f);
            }
            if(btn.classList.contains('mat-view-label')) btn.textContent = open ? 'Ocultar' : 'Visualizar';
          });
        })();
